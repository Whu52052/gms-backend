🌐 Nginx 负载均衡配置
upstream yunwei_backend { least_conn; server 10.5.50.30:8765 weight=3 max_fails=3 fail_timeout=30s; server 10.5.50.35:8765 weight=2 max_fails=3 fail_timeout=30s; keepalive 256; } server { listen 80; server_name _; location /api/ { proxy_pass http://yunwei_backend; proxy_http_version 1.1; proxy_set_header Connection ""; proxy_set_header Host $host; proxy_next_upstream error timeout http_502 http_503; } location /api/events { proxy_pass http://yunwei_backend; proxy_http_version 1.1; proxy_buffering off; proxy_read_timeout 86400s; proxy_next_upstream off; } location /ws { proxy_pass http://yunwei_backend; proxy_http_version 1.1; proxy_set_header Upgrade $http_upgrade; proxy_set_header Connection "upgrade"; } location / { proxy_pass http://yunwei_backend; } }
📜 自动部署脚本（复制到服务器执行）
#!/bin/bash # ================================ # 手套管理系统 - 一键部署脚本 # ================================ set -e echo "🚀 开始部署手套管理系统..." # 1. 安装依赖 echo "📦 安装 Node.js 和 npm..." curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash - sudo apt-get install -y nodejs # 2. 安装 PM2 echo "⚡ 安装 PM2 进程管理器..." sudo npm install -g pm2 # 3. 创建应用目录 echo "📁 创建应用目录..." sudo mkdir -p /opt/glove-management cd /opt/glove-management # 4. 复制代码（需要先上传代码包） # scp glove-management.tar.gz root@192.168.1.233:/opt/ # tar -xzf glove-management.tar.gz # 5. 安装依赖 echo "📦 安装应用依赖..." npm install # 6. 创建 .env 文件 echo "⚙️ 创建配置文件..." cat > .env << 'EOF' PORT=8765 SERVER_ROLE=primary SERVER_ID=server-001 DB_HOST=192.168.1.233 DB_PORT=3306 DB_USER=gms_user DB_PASSWORD=your_password DB_NAME=gms REDIS_URL=redis://192.168.1.233:6379 EOF # 7. 启动服务 echo "🚀 启动服务..." pm2 start ecosystem.config.js --name "glove-management" pm2 save pm2 startup # 8. 配置防火墙 echo "🔥 配置防火墙..." sudo ufw allow 8765/tcp sudo ufw allow 80/tcp sudo ufw allow 443/tcp echo "✅ 部署完成！" echo "访问地址: http://$(hostname -I | awk '{print $1}'):8765"
